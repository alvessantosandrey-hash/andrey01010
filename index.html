// --- Inimigos --- const enemies = []; function createEnemy(x, z) { const enemy = new THREE.Mesh( new THREE.BoxGeometry(1, 2, 1), new THREE.MeshStandardMaterial({ color: 0x0000ff }) ); enemy.position.set(x, 1, z); enemy.castShadow = true; scene.add(enemy); // Movimento aleatório enemy.userData.direction = new THREE.Vector3( (Math.random() - 0.5) * 0.05, 0, (Math.random() - 0.5) * 0.05 ); enemies.push(enemy); } // Criar alguns inimigos for (let i = 0; i < 5; i++) { createEnemy(Math.random() * 30 - 15, Math.random() * 30 - 15); } // --- Atualizar inimigos --- function updateEnemies() { enemies.forEach((enemy) => { enemy.position.add(enemy.userData.direction); // Limites da arena if (Math.abs(enemy.position.x) > 40 || Math.abs(enemy.position.z) > 40) { enemy.userData.direction.negate(); } }); } // --- Colisão bolas com inimigos --- function checkCollisions() { for (let i = balls.length - 1; i >= 0; i--) { const b = balls[i]; enemies.forEach((enemy, j) => { const dist = b.position.distanceTo(enemy.position); if (dist < 1) { // Inimigo atingido scene.remove(enemy); enemies.splice(j, 1); // Mancha de tinta no chão createSplat(enemy.position, b.material.color); // Remover bola scene.remove(b); balls.splice(i, 1); } }); } } // --- Loop Principal --- function animate() { requestAnimationFrame(animate); const delta = clock.getDelta(); // Movimento jogador const speed = 5 * delta; if (moveForward) camera.translateZ(-speed); if (moveBackward) camera.translateZ(speed); if (moveLeft) camera.translateX(-speed); if (moveRight) camera.translateX(speed); camera.position.y = 1.6; // Atualizar bolas balls.forEach((b, i) => { b.position.addScaledVector(b.userData.velocity, delta); b.userData.velocity.y -= 9.8 * delta; if (b.position.y <= 0) { createSplat(b.position, b.material.color); scene.remove(b); balls.splice(i, 1); } }); // Atualizar inimigos e colisões updateEnemies(); checkCollisions(); renderer.render(scene, camera); }
